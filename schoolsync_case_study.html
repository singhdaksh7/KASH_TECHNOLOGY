Multi-Tenant Architecture

Implementing a shared database with schema-level or row-level isolation. Each tenant has a unique identifier, and every query is strictly scoped to ensure data privacy. 

const tenantMiddleware = async (req, res, next) => {
  const hostname = req.headers['x-tenant-id'];
  const tenant = await prisma.tenant.findUnique({
    where: { subdomain: hostname }
  });
  if (!tenant) return res.status(404).json({ error: 'Tenant not found' });
  req.tenantId = tenant.id;
  next();
};
